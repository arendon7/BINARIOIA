  function mappedTimeAfterCuts(time, ranges) { let shift=0; for(const r of ranges){ if(time>=r.end) shift += r.end-r.start; else if(time>r.start) return r.start-shift; else break; } return Math.max(0,time-shift); }
  function timeInsideCuts(time, ranges) { return ranges.some(r => time>=r.start && time<r.end); }
  function normalizeCutRanges(rows, maxDuration) { const sorted=(rows||[]).map(r=>({start:clamp(Number(r.start),0,maxDuration),end:clamp(Number(r.end),0,maxDuration)})).filter(r=>r.end-r.start>.01).sort((a,b)=>a.start-b.start); const out=[]; for(const r of sorted){ const last=out[out.length-1]; if(last && r.start<=last.end+.001) last.end=Math.max(last.end,r.end); else out.push({...r}); } return out; }
  function splitItemAroundCuts(item, ranges) { const bounds=[item.start,item.start+item.duration]; const kept=[]; let cursor=bounds[0]; for(const r of ranges){ if(r.end<=cursor||r.start>=bounds[1]) continue; if(r.start>cursor) kept.push([cursor,Math.min(r.start,bounds[1])]); cursor=Math.max(cursor,r.end); if(cursor>=bounds[1]) break; } if(cursor<bounds[1]) kept.push([cursor,bounds[1]]); return kept.filter(([a,b])=>b-a>=.05).map(([a,b],idx)=>{ const c=clone(item); if(idx)c.id=uid("item"); c.start=mappedTimeAfterCuts(a,ranges); c.duration=b-a; c.source_in=Math.max(0,Number(item.source_in||0)+(a-item.start)); c.keyframes=(item.keyframes||[]).filter(k=>{const abs=item.start+Number(k.time||0);return abs>=a&&abs<=b;}).map(k=>({...k,time:Number((item.start+Number(k.time||0)-a).toFixed(3))})); return c; }); }
  function applyRippleCuts(ranges) { ranges=normalizeCutRanges(ranges,seq().duration); if(!ranges.length)return false; snapshot(); const oldDuration=seq().duration, oldPlayhead=seq().playhead; for(const t of seq().tracks){ const next=[]; for(const item of t.items) next.push(...splitItemAroundCuts(item,ranges)); t.items=next; } const tr=transcript(); tr.segments=(tr.segments||[]).flatMap(s=>{ const pseudo={id:s.id,track:"subtitles",start:Number(s.start),duration:Math.max(.05,Number(s.end)-Number(s.start)),source_in:0,keyframes:[]}; return splitItemAroundCuts(pseudo,ranges).map((x,idx)=>({...s,id:idx?uid("seg"):s.id,start:x.start,end:x.start+x.duration})); }); syncTranscriptToSubtitles(); const scenes=sceneStore(); scenes.scenes=(scenes.scenes||[]).filter(sc=>!timeInsideCuts(Number(sc.time),ranges)).map(sc=>({...sc,time:mappedTimeAfterCuts(Number(sc.time),ranges)})); seq().duration=Math.max(.1,oldDuration-ranges.reduce((a,r)=>a+(r.end-r.start),0)); seq().playhead=clamp(mappedTimeAfterCuts(oldPlayhead,ranges),0,seq().duration); silenceStore().applied_at=new Date().toISOString(); silenceStore().last_applied_ranges=clone(ranges); silenceStore().ranges=[]; silenceStore().total_silence=0; silenceStore().analyzed=false; selected=null; renderAll(); dirty(); return true; }
