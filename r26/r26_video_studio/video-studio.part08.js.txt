  $("fpsSelect").onchange = e => { snapshot(); state.settings.export.fps = Number(e.target.value); }; $("qualitySelect").onchange = e => { snapshot(); state.settings.export.quality_label = e.target.value; };
  document.addEventListener("keydown", e => {
    if (["INPUT", "SELECT", "TEXTAREA"].includes(document.activeElement.tagName)) return;
    if (e.code === "Space") { e.preventDefault(); togglePlay(); }
    if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "z") { e.preventDefault(); e.shiftKey ? redo() : undo(); }
    if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "y") { e.preventDefault(); redo(); }
    if (["Delete", "Backspace"].includes(e.key)) { e.preventDefault(); removeSelected(); }
    if (e.key.toLowerCase() === "s" && !e.metaKey && !e.ctrlKey) { e.preventDefault(); splitSelected(); }
    if (e.key.toLowerCase() === "k" && !e.metaKey && !e.ctrlKey) { e.preventDefault(); addKeyframe(); }
    if (e.key.toLowerCase() === "v" && !e.metaKey && !e.ctrlKey) { e.preventDefault(); addAudioKeyframe(); }
    if (e.key.toLowerCase() === "c" && !e.metaKey && !e.ctrlKey) { e.preventDefault(); addSubtitleSample(); }
    if (e.key.toLowerCase() === "f" && !e.metaKey && !e.ctrlKey) { e.preventDefault(); snapshot(); state.settings.focus_mode=!state.settings.focus_mode; syncControls(); }
    if (e.key === "0") { e.preventDefault(); fitTimeline(); }
    if (e.key === "ArrowLeft") { e.preventDefault(); setPlayhead(seq().playhead - (e.shiftKey ? 5 : 1)); }
    if (e.key === "ArrowRight") { e.preventDefault(); setPlayhead(seq().playhead + (e.shiftKey ? 5 : 1)); }
  });

    await restoreEditingMode();
    await attachRequestedProject();
    renderAll(); syncProjectLocation();
    await ensureProjectStorage(); savePhysical();
    refreshEngineStatus();
  }
  bootR27VideoStudio();
