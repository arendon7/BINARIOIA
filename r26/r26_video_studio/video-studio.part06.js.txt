  function createClipFromRange(start, end, name = "Clip") {
    const src = state.sequences.find(s => s.kind === "long") || seq(); const c = clone(src); c.id = uid("seq"); c.kind = "clip"; c.name = name || `Clip ${state.sequences.filter(s => s.kind === "clip").length + 1}`; c.duration = Math.max(.2, end - start); c.playhead = 0;

  }
