  function currentItemAt(trackKind) { const t = track(trackKind); if (!t || !t.enabled) return null; return [...t.items].reverse().find(i => i.enabled && seq().playhead >= i.start && seq().playhead < i.start + i.duration); }
  function interpolate(a, b, p) { return a + (b - a) * p; }
  function effectiveTransform(item) {
    ensureItemShape(item); const local = seq().playhead - item.start; const frames = [...item.keyframes].sort((a, b) => a.time - b.time); if (!frames.length) return item.transform;
    let prev = {time: 0, transform: item.transform}, next = null;
    for (const f of frames) { if (f.time <= local) prev = f; else { next = f; break; } }
    if (!next) return prev.transform || item.transform;
    const p = clamp((local - prev.time) / Math.max(.001, next.time - prev.time), 0, 1), at = prev.transform || item.transform, bt = next.transform || at;
    const crop = {}; for (const side of ["top", "right", "bottom", "left"]) crop[side] = interpolate(Number(at.crop?.[side] || 0), Number(bt.crop?.[side] || 0), p);
    return {x: interpolate(Number(at.x ?? .5), Number(bt.x ?? .5), p), y: interpolate(Number(at.y ?? .5), Number(bt.y ?? .5), p), scale: interpolate(Number(at.scale ?? 1), Number(bt.scale ?? 1), p), rotation: interpolate(Number(at.rotation || 0), Number(bt.rotation || 0), p), crop, fit: p < .5 ? (at.fit || "contain") : (bt.fit || "contain")};
  }
  function transitionVisual(item, transform) {
    const local = clamp(seq().playhead - item.start, 0, item.duration), remaining = Math.max(0, item.duration - local);
    let alpha = Number(item.opacity ?? 1), x = Number(transform.x ?? .5), y = Number(transform.y ?? .5), scale = Number(transform.scale ?? 1);
    const apply = (tr, progress, entering) => {
      const type = tr?.type || "none"; if (type === "none") return;
      const p = clamp(progress, 0, 1), q = entering ? p : 1 - p;
      if (["fade", "dissolve"].includes(type)) alpha *= q;
      if (type === "slide_left") x += (entering ? -0.18 * (1 - p) : -0.18 * p);
      if (type === "slide_right") x += (entering ? 0.18 * (1 - p) : 0.18 * p);
      if (type === "zoom") scale *= entering ? (0.82 + 0.18 * p) : (1 - 0.18 * p);
    };
    const tin = item.transitions?.in, tout = item.transitions?.out;
    if (tin?.duration > 0 && local < tin.duration) apply(tin, local / tin.duration, true);
    if (tout?.duration > 0 && remaining < tout.duration) apply(tout, remaining / tout.duration, false);
    return {...transform, x, y, scale, opacity: clamp(alpha, 0, 1)};
  }

  function setMediaSource(el, asset) {
    const url = asset ? mediaUrl(asset) : ""; if (!url) { el.classList.add("hidden"); el.removeAttribute("src"); return; }
    if (el.dataset.assetId !== asset.id) { el.dataset.assetId = asset.id; el.src = url; }
    el.classList.remove("hidden");
  }

  }
  function colorCssFilter(item) { ensureItemShape(item); const c=item.color; const bright=Math.max(.05,1+Number(c.brightness||0)); const contrast=Math.max(0,Number(c.contrast??1)); const saturation=Math.max(0,Number(c.saturation??1)); const gamma=Math.max(.1,Number(c.gamma??1)); const temp=Number(c.temperature||0), tint=Number(c.tint||0); const hue=(tint*8)+(temp*3); const sepia=Math.abs(temp)*.12, gammaBright=Math.pow(1/gamma,.22); return `brightness(${bright*gammaBright}) contrast(${contrast}) saturate(${saturation}) sepia(${sepia}) hue-rotate(${hue}deg)`; }
