  function wireTransformBox() {
    const box = $("transformBox");
    box.onpointerdown = e => {
      if (!selected) return; const [, item] = findItem(selected); if (!item || !VISUAL_TRACKS.has(item.track)) return;
      const rect = $("videoCanvas").getBoundingClientRect(), base = clone(item.transform), startX = e.clientX, startY = e.clientY, scaling = e.target.classList.contains("transform-handle");
      snapshot(); box.setPointerCapture(e.pointerId); e.preventDefault(); e.stopPropagation();
      const move = ev => {
        const dx = ev.clientX - startX, dy = ev.clientY - startY;
        if (scaling) item.transform.scale = clamp(Number(base.scale || 1) + (dx / Math.max(1, rect.width) + dy / Math.max(1, rect.height)), .05, 20);
        else { item.transform.x = clamp(Number(base.x ?? .5) + dx / Math.max(1, rect.width), -2, 3); item.transform.y = clamp(Number(base.y ?? .5) + dy / Math.max(1, rect.height), -2, 3); }
        renderPreview(); renderInspector();
      };
      const up = () => { document.removeEventListener("pointermove", move); document.removeEventListener("pointerup", up); dirty(); };
      document.addEventListener("pointermove", move); document.addEventListener("pointerup", up);
    };
  }
