  function waveformHtml(item) {
    if (!AUDIO_TRACKS.has(item.track) && !["video", "broll"].includes(item.track)) return "";
    const asset = state.assets.find(a => a.id === item.asset_id), peaks = asset?.metadata?.waveform;
    if (!Array.isArray(peaks) || !peaks.length) return "";
    const count = Math.max(12, Math.min(80, Math.floor(item.duration * pxSec / 4)));
    const stride = Math.max(1, Math.floor(peaks.length / count));
    const sampled = []; for (let i = 0; i < peaks.length && sampled.length < count; i += stride) sampled.push(peaks[i]);
    return `<div class="waveform">${sampled.map(v => `<i style="height:${Math.max(3, Math.round(v * 32))}px"></i>`).join("")}</div>`;
  }
  function audioEnvelopeHtml(item) {
    if (!AUDIO_CAPABLE_TRACKS.has(item.track)) return "";
    const points = [{time:0, volume:Number(item.volume ?? 1)}, ...(item.keyframes || []).filter(k => k.volume != null).map(k => ({time:Number(k.time||0), volume:Number(k.volume)}))].sort((a,b)=>a.time-b.time);
    if (points.length < 2) return "";
    const coords = points.map(p => `${clamp(p.time / Math.max(.001,item.duration),0,1)*100},${100-clamp(p.volume/4,0,1)*100}`).join(" ");
    return `<svg class="audio-envelope" viewBox="0 0 100 100" preserveAspectRatio="none"><polyline points="${coords}"></polyline></svg>`;
  }

  function timelineMode() { return state.settings.timeline_edit_mode || "magnetic"; }
  function localMagnetStart(item, desired, threshold = .18) {
    if (timelineMode() === "free" || !$("snapToggle")?.checked) return Math.max(0, desired);
    const points = [0, seq().duration];
    for (const t of seq().tracks) for (const other of t.items) if (other.id !== item.id) { points.push(Number(other.start||0), Number(other.start||0)+Number(other.duration||0)); }
    if (seq().kind === "long") for (const scene of (sceneStore().scenes || [])) points.push(Number(scene.time||0));
    let best = desired, dist = threshold + 1;
    for (const p of points) {
      const d1 = Math.abs(desired - p); if (d1 < dist && d1 <= threshold) { best = p; dist = d1; }
      const endStart = p - Number(item.duration||0), d2 = Math.abs(desired - endStart); if (d2 < dist && d2 <= threshold) { best = endStart; dist = d2; }
    }
    return Math.max(0, best);
  }
  async function commitTimelineMove(baseProject, itemId, requestedStart) {
    try {
      const r = await apiJson("/api/timeline/move", {method:"POST", headers:{"Content-Type":"application/json"}, body:JSON.stringify({project:baseProject, sequence_id:baseProject.active_sequence_id, item_id:itemId, new_start:requestedStart, mode:timelineMode(), threshold:.18, ripple_scope:"all"})});
      state = r.project; ensureShape(); selected = itemId; renderAll(); dirty();
      if (r.snapped_to != null) toast("Ajuste magnético aplicado");
      if (r.mode === "ripple" && r.shifted_items) toast(`Ripple: ${r.shifted_items} elementos desplazados`);
    } catch (e) { state = baseProject; ensureShape(); renderAll(); toast(`No pude mover el elemento: ${e.message}`, true); }
  }
  function wireItem(el) {
    el.onclick = e => { e.stopPropagation(); selected = el.dataset.item; renderAll(); };
    el.onpointerdown = e => {
      if (e.button !== 0) return; const [t, i] = findItem(el.dataset.item); if (!i || t.locked) return; selected = i.id;
      const handle = e.target.classList.contains("resize-handle") ? (e.target.classList.contains("left") ? "left" : "right") : "move";
      const startX = e.clientX, baseStart = i.start, baseDur = i.duration, baseSource = i.source_in || 0, dragBase = clone(state); snapshot(); el.setPointerCapture(e.pointerId);
      const move = ev => {
        const dx = (ev.clientX - startX) / pxSec; let ns = baseStart, nd = baseDur, source = baseSource;
        if (handle === "move") ns = localMagnetStart(i, baseStart + dx);
        if (handle === "left") { ns = baseStart + dx; nd = baseDur - dx; source = Math.max(0, baseSource + dx); }
        if (handle === "right") nd = baseDur + dx;
        if ($("snapToggle").checked && handle !== "move") { const snap = .25; ns = Math.round(ns / snap) * snap; nd = Math.round(nd / snap) * snap; }
        ns = clamp(ns, 0, Math.max(.05, seq().duration - .05)); nd = clamp(nd, .05, Math.max(.05, seq().duration - ns)); i.start = ns; i.duration = nd; i.source_in = source;
        el.classList.toggle("magnetic-snap", handle === "move" && timelineMode() === "magnetic" && Math.abs(ns-(baseStart+dx))>.001);
        el.classList.toggle("ripple-edit", handle === "move" && timelineMode() === "ripple");
        renderTimeline(); renderInspector(); renderPreview();
      };
      const up = () => { document.removeEventListener("pointermove", move); document.removeEventListener("pointerup", up); const finalStart = i.start; if (handle === "move") commitTimelineMove(dragBase, i.id, finalStart); else dirty(); };
      document.addEventListener("pointermove", move); document.addEventListener("pointerup", up);
    };
  }
