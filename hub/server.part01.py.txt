def process_env(app_dir=None):
    env=os.environ.copy()
    try:
        runtime_venv=(ROOT/".venv").resolve(strict=True)
        runtime_root=runtime_venv.parent
    except Exception:
        runtime_venv=ROOT/".venv"
        runtime_root=runtime_venv.parent
    runtime_bin=str(runtime_root/"bin")
    paths=[runtime_bin,str(runtime_venv/"bin"),str(ROOT),"/opt/homebrew/bin","/usr/local/bin",env.get("PATH","")]
    if app_dir: paths.insert(0,str(app_dir))
    env["PATH"]=":".join(x for x in paths if x)
    env["PYTHONPATH"]=":".join([str(ROOT),str(app_dir) if app_dir else "",env.get("PYTHONPATH","")])
    env["BINARIO_IA_ROOT"]=str(ROOT)
    env["BINARIO_AI_CONFIG_ROOT"]=str(Path.home()/"Library"/"Application Support"/"Binario IA"/"ai")
    env["BINARIO_WORKSPACES_HOME"]=os.environ.get("BINARIO_WORKSPACES_HOME",str(Path.home()/"Documents"/"Binario IA"/"Workspaces"))
    try:
        from runtime.runtime_manager import _known_path
        f=_known_path("ffmpeg"); p=_known_path("ffprobe")
        if f: env["BINARIO_FFMPEG"]=f
        if p: env["BINARIO_FFPROBE"]=p
    except Exception: pass
    if os.environ.get("BINARIO_HUB_URL"): env["BINARIO_HUB_URL"]=os.environ["BINARIO_HUB_URL"]
    return env

def service_id_for(app):
    return f"binario-v0251a1-app-{app['id']}"

def _service_root_ok(port:int, app:dict)->bool:
    """Validate both service identity and the browser entrypoint before reuse/navigation."""
    sid=service_id_for(app)
    if not service_matches(port,sid):
        return False
    try:
        req=Request(f"http://127.0.0.1:{int(port)}/",headers={"Cache-Control":"no-cache","Accept":"text/html"})
        with urlopen(req,timeout=.8) as r:
            if r.status!=200:
                return False
            text=r.read(12000).decode("utf-8","ignore")
        if app["id"]=="05-editor-video-ia":
            markers=["Editor Video IA", "Sistema Binario · Editor Video"]
        elif app["id"]=="01-auditoria-negocio":
            markers=["Audit Studio","Auditoría de Negocio"]
        elif app["id"]=="02-cazador-webs":
            markers=["Web Studio","Cazador de Webs"]
        elif app["id"]=="03-agente-ia-whatsapp":
            markers=["Agent Studio","WhatsApp"]
        elif app["id"]=="04-auditoria-youtube":
            markers=["YouTube Studio","Content Intelligence"]
        elif app["id"]=="06-auditoria-marca-personal":
            markers=["Brand Studio","Content Intelligence"]
        elif app["id"]=="07-auditoria-ecommerce":
            markers=["Commerce Studio","Auditoría E-commerce"]
        elif app["id"]=="09-propuestas-ia":
            markers=["Proposal Studio","Propuestas IA"]
        elif app["id"]=="11-documentos-ia":
            markers=["Documentos IA"]
        elif app["id"]=="12-app-factory-ia":
            markers=["Sistema Binario · App Factory","App Factory OS","Project Intelligence Hub"]
        else:
            markers=[app.get("name","")]
        low=text.lower()
        return any(m and m.lower() in low for m in markers)
    except Exception:
        return False
