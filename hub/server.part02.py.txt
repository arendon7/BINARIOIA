def start_runtime_repair(scope="essential"):
    with RUNTIME_JOB_LOCK:
        if RUNTIME_JOB.get("running"):
            return {"status":"already_running","job":dict(RUNTIME_JOB)}
        RUNTIME_JOB.update({"running":True,"progress":0.01,"message":"Iniciando reparación…","result":None,"error":None,"scope":scope})
    def worker():
        try:
            from runtime.runtime_manager import repair
            def progress(p,msg):
                with RUNTIME_JOB_LOCK:RUNTIME_JOB.update({"progress":float(p),"message":str(msg)})
            result=repair(scope,ROOT,progress)
            with RUNTIME_JOB_LOCK:RUNTIME_JOB.update({"running":False,"progress":1.0,"message":"Runtime listo." if result.get("ok") else "La reparación terminó con pendientes.","result":result,"error":None})
        except Exception as exc:
            with RUNTIME_JOB_LOCK:RUNTIME_JOB.update({"running":False,"message":"Falló la reparación.","error":f"{type(exc).__name__}: {exc}"})
    threading.Thread(target=worker,daemon=True).start()
    return {"status":"started","job":dict(RUNTIME_JOB)}

def execute_workflow(recipe, initial):
    data=initial; trace=[]
    for idx,step in enumerate(recipe.get("steps",[]),1):
        app=app_by_id(step.get("app_id",""))
        if not app: raise RuntimeError(f"App desconocida en workflow: {step.get('app_id')}")
        if app.get("special_service"):
            trace.append({"step":idx,"app":app["id"],"status":"manual_service","url":launch_service(app)})
            continue
        payload=dict(step.get("defaults") or {})
        if step.get("pass_previous",True) and isinstance(data,dict):
            payload.setdefault("previous",data)
        payload.update(initial.get(app["id"],{}) if isinstance(initial.get(app["id"]),dict) else {})
        out=load_engine(app).run(payload)
        trace.append({"step":idx,"app":app["id"],"status":out.get("status"),"summary":out.get("summary"),"paths":out.get("paths")})
        data=out
        if out.get("status")=="blocked": break
    return {"status":"pass" if trace and trace[-1]["status"]!="blocked" else "blocked","trace":trace,"last":data}

class Handler(SimpleHTTPRequestHandler):
    def log_message(self,fmt,*args): pass
    def _json(self,data,status=200):
        raw=json.dumps(data,ensure_ascii=False,indent=2,default=str).encode("utf-8")
        self.send_response(status); self.send_header("Content-Type","application/json; charset=utf-8")
        self.send_header("Content-Length",str(len(raw))); self.end_headers(); self.wfile.write(raw)
    def _body(self):
        n=int(self.headers.get("Content-Length","0") or 0)
        raw=self.rfile.read(n) if n else b"{}"
        return json.loads(raw or b"{}")
    def _file(self,p):
        p=Path(p)
        if not p.exists() or not p.is_file():
            self.send_error(404); return
        data=p.read_bytes(); c=mimetypes.guess_type(str(p))[0] or "application/octet-stream"
        self.send_response(200); self.send_header("Content-Type",c); self.send_header("Content-Length",str(len(data))); self.send_header("Cache-Control","no-cache"); self.end_headers(); self.wfile.write(data)
    def do_GET(self):
        parsed=urlparse(self.path)
        path=unquote(parsed.path)
